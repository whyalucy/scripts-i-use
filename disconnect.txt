-- // 3c (Speed FFlags) //
local function applySpeedFlags()
	local flags = {
		DFIntCharacterMaxSpeed = "9999",
		DFIntMaxLinearVelocity = "9999",
		DFIntTaskSchedulerTargetFps = "240",
		DFIntPhysicsSenderRate = "240",
		PhysicsSenderMaxBandwidthBps = "20000",
		ServerMaxBandwith = "9999",
		MaxDataPacketPerSend = "2147483647",
		InterpolationFrameVelocityThresholdMillionth = "1",
		InterpolationFramePositionThresholdMillionth = "1",
	}
	for key, value in pairs(flags) do
		pcall(function() setfflag(key, value) end)
	end
end

-- // 3d (PC raknet hook) //
local hooked = false
local function rakhook(packet)
	if packet.PacketId == 0x1B then
		local data = packet.AsBuffer
		buffer.writeu32(data, 1, 0xFFFFFFFF)
		packet:SetData(data)
	end
end

local function enablePCDesync()
	if not hooked then
		pcall(function()
			raknet.add_send_hook(rakhook)
			hooked = true
		end)
	end
end

local function disablePCDesync()
	if hooked then
		pcall(function()
			raknet.remove_send_hook(rakhook)
			hooked = false
		end)
	end
end

-- // 3e (Mobile desync) //
local function enableMobileDesync()
	applySpeedFlags()
	task.wait(0.4)
	pcall(function() raknet.desync(true) end)
end

local function disableMobileDesync()
	pcall(function() raknet.desync(false) end)
end

-- // 18 (Draggable via header) //
do
	local dg, ds, sp = false, nil, nil
	local function startDrag(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dg = true; ds = input.Position; sp = main.Position
		end
	end
	collapseBtn.InputBegan:Connect(startDrag)
	titleLbl.InputBegan:Connect(startDrag)

	UserInputService.InputChanged:Connect(function(input)
		if dg and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - ds
			main.Position = UDim2.new(sp.X.Scale, sp.X.Offset + delta.X, sp.Y.Scale, sp.Y.Offset + delta.Y)
		end
	end)

	local function endDrag(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dg = false
		end
	end
	collapseBtn.InputEnded:Connect(endDrag)
	titleLbl.InputEnded:Connect(endDrag)
end

-- // 19 (Desync logic — raknet, platform-smart) //
local function startDesync()
	desyncEnabled = true
	applySpeedFlags()
	task.wait(0.3)
	if isMobile then
		enableMobileDesync()
	else
		enablePCDesync()
	end
end

local function stopDesync()
	desyncEnabled = false
	if isMobile then
		disableMobileDesync()
	else
		disablePCDesync()
	end
end

desyncBtn.MouseButton1Click:Connect(function()
	if desyncEnabled then
		stopDesync()
		desyncBtn.Text = "TOGGLE DESYNC"
		statusValue.Text = "OFF"
		statusValue.TextColor3 = Color3.fromRGB(200, 80, 80)
		mainStroke.Color = Color3.fromRGB(210, 210, 210)
	else
		startDesync()
		desyncBtn.Text = "DESYNC ACTIVE"
		statusValue.Text = "ACTIVE"
		statusValue.TextColor3 = Color3.fromRGB(80, 200, 80)
		mainStroke.Color = Color3.fromRGB(145, 144, 144)
	end
end)

-- // 20 (No Anim toggle) //
local savedAnimate = nil

noAnimBtn.MouseButton1Click:Connect(function()
	noAnimEnabled = not noAnimEnabled
	noAnimBtn.Text = noAnimEnabled and "NO ANIM: ON" or "NO ANIM: OFF"

	pcall(function()
		local char = lp.Character
		if not char then return end
		if noAnimEnabled then
			local animate = char:FindFirstChild("Animate")
			if animate then
				savedAnimate = animate:Clone()
				animate:Destroy()
			end
		else
			if savedAnimate and lp.Character then
				savedAnimate.Parent = lp.Character
				savedAnimate = nil
			end
		end
	end)
end)

-- // 21 (Re-apply on respawn) //
lp.CharacterAdded:Connect(function(char)
	task.wait(1)
	if desyncEnabled then
		stopDesync()
		task.wait(0.5)
		startDesync()
	end
	if noAnimEnabled then
		local animate = char:FindFirstChild("Animate")
		if animate then
			savedAnimate = animate:Clone()
			animate:Destroy()
		end
	end
end)

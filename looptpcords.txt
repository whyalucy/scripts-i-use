    end

menuBtn.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        menuDragToggle = false
        -- Reset color when dragging ends
        menuBtn.BackgroundColor3 = Color3.fromRGB(50, 150, 255)
    end
end)

menuBtn.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
        menuDragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == menuDragInput and menuDragToggle then
        local delta = input.Position - menuDragStart
        local newX = menuStartPos.X.Offset + delta.X
        local newY = menuStartPos.Y.Offset + delta.Y
        
        -- Keep button within screen bounds
        local screenSize = gui.AbsoluteSize
        newX = math.max(0, math.min(newX, screenSize.X - menuBtn.AbsoluteSize.X))
        newY = math.max(0, math.min(newY, screenSize.Y - menuBtn.AbsoluteSize.Y))
        
        menuBtn.Position = UDim2.new(0, newX, 0, newY)
    end
end)

menuBtn.MouseButton1Click:Connect(function()
    mainFrame.Visible = not mainFrame.Visible
    print("Menu toggled:", mainFrame.Visible and "OPENED" or "CLOSED")

-- Copy to clipboard function
local function copyToClipboard(text)
    pcall(function()
        local clipboard = game:GetService("ClipboardService")
        clipboard:Set(text)
        print("Copied to clipboard:", text)
    end)
end

-- Show copyable link function
local function showCopyableLink(message, link)
    local notification = Instance.new("Frame", gui)
    notification.Size = UDim2.new(0, 400, 0, 120)
    notification.Position = UDim2.new(0.5, -200, 0, -120)
    notification.BackgroundColor3 = Color3.fromRGB(0, 100, 200)
    notification.BorderSizePixel = 0
    
    local notifCorner = Instance.new("UICorner", notification)
    notifCorner.CornerRadius = UDim.new(0, 10)
    
    local notifText = Instance.new("TextLabel", notification)
    notifText.Size = UDim2.new(1, 0, 0, 30)
    notifText.Position = UDim2.new(0, 0, 0, 0)
    notifText.BackgroundTransparency = 1
    notifText.Text = message
    notifText.TextColor3 = Color3.new(1, 1, 1)
    notifText.TextSize = 16
    notifText.Font = Enum.Font.SourceSansBold
    notifText.TextXAlignment = Enum.TextXAlignment.Center
    notifText.TextYAlignment = Enum.TextYAlignment.Center
    
    -- Copyable TextBox
    local linkBox = Instance.new("TextBox", notification)
    linkBox.Size = UDim2.new(1, -20, 0, 40)
    linkBox.Position = UDim2.new(0, 10, 0, 40)
    linkBox.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    linkBox.Text = link
    linkBox.TextColor3 = Color3.new(0, 0, 0)
    linkBox.TextSize = 12
    linkBox.Font = Enum.Font.SourceSans
    linkBox.TextXAlignment = Enum.TextXAlignment.Center
    linkBox.TextYAlignment = Enum.TextYAlignment.Center
    linkBox.BorderSizePixel = 0
    linkBox.ClearTextOnFocus = false  -- This is the key setting!
    linkBox.TextEditable = false      -- Prevent editing
    
    local linkCorner = Instance.new("UICorner", linkBox)
    linkCorner.CornerRadius = UDim.new(0, 5)
    
    local instructionText = Instance.new("TextLabel", notification)
    instructionText.Size = UDim2.new(1, 0, 0, 30)
    instructionText.Position = UDim2.new(0, 0, 0, 90)
    instructionText.BackgroundTransparency = 1
    instructionText.Text = "Click on the link above to select and copy it"
    instructionText.TextColor3 = Color3.fromRGB(200, 200, 200)
    instructionText.TextSize = 12
    instructionText.Font = Enum.Font.SourceSans
    instructionText.TextXAlignment = Enum.TextXAlignment.Center
    instructionText.TextYAlignment = Enum.TextYAlignment.Center
    
    -- Animate notification in
    notification:TweenPosition(UDim2.new(0.5, -200, 0, 20), "Out", "Quad", 0.5)
    
    -- Remove notification after 8 seconds
    task.spawn(function()
        task.wait(8)
        notification:TweenPosition(UDim2.new(0.5, -200, 0, -120), "Out", "Quad", 0.5)
        task.wait(0.5)
        notification:Destroy()
    end)
end

-- Auto Teleport Loop
task.spawn(function()
    while scriptRunning do
        if isActive and scriptRunning then
            -- check character and HumanoidRootPart
            if character and character:FindFirstChild("HumanoidRootPart") then
                local x = tonumber(xBox.Text) or 0
                local y = tonumber(yBox.Text) or 0
                local z = tonumber(zBox.Text) or 0
                
                -- teleport
                pcall(function()
                    character.HumanoidRootPart.CFrame = CFrame.new(x, y, z)
                    print("Teleported to:", x, y, z)
                end)
            else
                print("Character or HumanoidRootPart not found!")
            end
        end
        local interval = tonumber(intervalBox.Text) or 1
        task.wait(interval)
    end
    print("Teleport loop stopped!")
end)

-- Auto Jump Loop
task.spawn(function()
    while scriptRunning do
        if isJumpActive and scriptRunning then
            -- check character and Humanoid
            if character and character:FindFirstChild("Humanoid") then
                -- jump
                pcall(function()
                    character.Humanoid.Jump = true
                    print("Jumped!")
                end)
            else
                print("Character or Humanoid not found!")
            end
        end
        local jumpInterval = tonumber(jumpIntervalBox.Text) or 0.5
        task.wait(jumpInterval)
    end
    print("Jump loop stopped!")
end)
